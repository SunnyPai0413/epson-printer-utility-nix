cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" EPSON_VERSION LIMIT_COUNT 1)
project(epson-native VERSION ${EPSON_VERSION} LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
option(BUILD_UTILITY "Build native GUI and CLI" ON)
option(BUILD_BACKEND "Build native ecbd and CUPS ecblp" ON)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# The upstream C sources predate GCC 14's stricter diagnostics.
add_compile_options($<$<COMPILE_LANGUAGE:C>:-Wno-error=implicit-function-declaration> $<$<COMPILE_LANGUAGE:C>:-Wno-error=incompatible-pointer-types>)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
include_directories(src)
add_compile_definitions(EPSON_PROJECT_VERSION="${PROJECT_VERSION}")
if(BUILD_BACKEND)
  pkg_check_modules(USB REQUIRED IMPORTED_TARGET libusb-1.0)
  set(B epson-backend)
  include_directories(${B}/src)
  add_executable(ecbd ${B}/daemon/cbtd.c ${B}/daemon/cbtd_comserv.c
    ${B}/daemon/cbtd_datatrans.c ${B}/daemon/cbtd_setup.c ${B}/daemon/cbtd_signal.c
    ${B}/daemon/cbtd_thread.c ${B}/daemon/cbtd_wrapper.c ${B}/daemon/ecbteg.c
    ${B}/daemon/daemon_osfunc.c ${B}/daemon/epmnt.c)
  target_compile_definitions(ecbd PRIVATE PACKAGE_STRING="epson-backend-1.3.1")
  target_link_libraries(ecbd PRIVATE PkgConfig::USB Threads::Threads)
  add_executable(ecblp ${B}/src/backend.c ${B}/src/str.c ${B}/src/err.c ${B}/src/mem.c)
  target_compile_definitions(ecblp PRIVATE PACKAGE_STRING="epson-backend-1.3.1")
  install(TARGETS ecbd RUNTIME DESTINATION bin)
  install(TARGETS ecblp RUNTIME DESTINATION lib/cups/backend)
  if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(ecblp-fixture ${B}/src/backend.c ${B}/src/str.c ${B}/src/err.c ${B}/src/mem.c)
    target_compile_definitions(ecblp-fixture PRIVATE PACKAGE_STRING="epson-backend-1.3.1" EPSON_RUNTIME="${CMAKE_CURRENT_BINARY_DIR}/cups-runtime")
    add_test(NAME cups-fifo COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cups.py $<TARGET_FILE:ecblp> $<TARGET_FILE:ecblp-fixture> ${CMAKE_CURRENT_BINARY_DIR}/cups-runtime)
  endif()
  install(FILES ${B}/COPYING ${B}/AUTHORS ${B}/README DESTINATION share/doc/epson-backend)
endif()
if(BUILD_UTILITY)
  find_package(Qt5 5.15 REQUIRED COMPONENTS Core Widgets Network Concurrent)
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  set(U epson-printer-utility)
  set(L ${U}/EPSCommonLib)
  # Follow precisely the sources listed in Epson's library Makefile.am.
  file(READ ${L}/Makefile.am upstream_makefile)
  string(REGEX MATCH "libEPSCommonLib_a_SOURCES =([\n\r]|.)*" upstream_sources "${upstream_makefile}")
  string(REGEX MATCHALL "[.A-Za-z0-9_/-]+\\.(cpp|c)" sources "${upstream_sources}")
  list(TRANSFORM sources PREPEND "${L}/")
  # Epson builds these .c files with g++; one embeds the C++ XAsyncSocket class.
  set_source_files_properties(${sources} PROPERTIES LANGUAGE CXX)
  add_library(EPSCommonLib STATIC ${sources} ${U}/EPUCommon/EThread.cpp)
  target_include_directories(EPSCommonLib PUBLIC ${L} ${L}/escpr-lib/inc ${L}/epsmp/linux ${L}/epsmp ${U})
  target_compile_definitions(EPSCommonLib PUBLIC QT_NO_DEBUG_OUTPUT GCOMSW_PRT_USE_LPR GCOMSW_PRT_USE_RAW GCOMSW_EL_CDLABEL GCOMSW_EF_MAINTE)
  target_link_libraries(EPSCommonLib PUBLIC Qt5::Core Qt5::Network Threads::Threads jpeg m)
  set(G ${U}/PrinterUtility)
  add_executable(epson-printer-utility ${G}/Main.cpp ${G}/EPUMainWindow.cpp
    ${G}/EPUMainController.cpp ${G}/EPUView.cpp ${G}/EPUMaintenanceGuideView.cpp
    ${G}/EPUMainView.cpp ${G}/EPUCallbackObject.cpp ${G}/EPUOperationView.cpp
    ${G}/EPUGlobal.cpp ${G}/EPUInkLevelView.cpp ${G}/EPUPrinterListDialog.cpp
    ${G}/EPUPrinterController.cpp ${G}/EPUPrinter.cpp ${G}/Resources.qrc)
  target_link_libraries(epson-printer-utility PRIVATE EPSCommonLib Qt5::Widgets Qt5::Concurrent)
  target_compile_definitions(epson-printer-utility PRIVATE EPSON_RESOURCE_DIR="${CMAKE_INSTALL_FULL_DATADIR}/epson-printer-utility/Languages")
  add_executable(epson-printer-cli src/cli.cpp)
  target_link_libraries(epson-printer-cli PRIVATE EPSCommonLib)
  install(TARGETS epson-printer-utility epson-printer-cli RUNTIME DESTINATION bin)
  install(DIRECTORY ${G}/Languages/ DESTINATION share/epson-printer-utility/Languages FILES_MATCHING PATTERN "*.qm")
  install(FILES src/epson-printer-utility.desktop DESTINATION share/applications)
  install(FILES ${G}/Images/AppIcon.png DESTINATION share/icons/hicolor/256x256/apps RENAME epson-printer-utility.png)
  install(FILES ${U}/COPYING ${U}/AUTHORS ${U}/README DESTINATION share/doc/epson-printer-utility)
  if(BUILD_TESTING)
    add_executable(report-test tests/report.cpp)
    target_link_libraries(report-test PRIVATE EPSCommonLib)
    add_test(NAME reported-model COMMAND report-test)
    add_executable(network-parser-test tests/network-parser.cpp)
    target_link_libraries(network-parser-test PRIVATE EPSCommonLib)
    add_test(NAME native-network-parser COMMAND network-parser-test)
    add_executable(parser-test tests/parser.cpp ${U}/BackendLib/backend_api.c)
    target_include_directories(parser-test PRIVATE ${U})
    add_test(NAME native-parser COMMAND parser-test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME upstream-modification-notices COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/notices.py ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME cli-offline COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/offline.py $<TARGET_FILE:epson-printer-cli>)
  endif()
endif()

install(FILES LICENSE NOTICE.md CHANGELOG.md UPSTREAM-MODIFICATIONS.txt DESTINATION share/doc/epson-printer-utility-nix)
install(DIRECTORY LICENSES/ DESTINATION share/doc/epson-printer-utility-nix/licenses)
